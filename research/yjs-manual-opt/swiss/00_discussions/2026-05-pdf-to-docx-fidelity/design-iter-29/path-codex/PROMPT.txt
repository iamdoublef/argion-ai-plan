import fitz, numpy as np
from PIL import Image
from pathlib import Path
def render_pdf_pngs(pdf, outdir):
    doc = fitz.open(pdf); outdir.mkdir(exist_ok=True, parents=True)
    for i,p in enumerate(doc):
        pix = p.get_pixmap(matrix=fitz.Matrix(150/72,150/72), alpha=False)
        pix.save(str(outdir / f'page-{i+1:02d}.png'))
def diff_dirs(a_dir, b_dir):
    a=sorted(Path(a_dir).glob('*.png')); b=sorted(Path(b_dir).glob('*.png'))
    ds=[]
    for x,y in zip(a,b):
        ai=np.array(Image.open(x).convert('RGB')); bi=np.array(Image.open(y).convert('RGB'))
        if ai.shape!=bi.shape:
            from PIL import Image as I
            bi=np.array(I.open(y).convert('RGB').resize((ai.shape[1],ai.shape[0])))
        ds.append(float(np.abs(ai.astype(int)-bi.astype(int)).mean()))
    return ds
